cmake_minimum_required(VERSION 3.16)
project(car_vision LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(CAR_BUILD_RKNN "Build board RKNN image detector" ON)
if(BUILD_TESTING)
  add_executable(postprocess_test tests/postprocess_test.cpp)
  target_include_directories(postprocess_test PRIVATE include)
  target_compile_options(postprocess_test PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME yolo_postprocess COMMAND postprocess_test)
  add_executable(postprocess_heads tests/postprocess_heads.cpp)
  target_include_directories(postprocess_heads PRIVATE include)
endif()
if(CAR_BUILD_RKNN)
  set(RKNN_INCLUDE_DIR "" CACHE PATH "Directory containing matching rknn_api.h")
  if(NOT EXISTS "${RKNN_INCLUDE_DIR}/rknn_api.h")
    message(FATAL_ERROR "Set RKNN_INCLUDE_DIR to the matching SDK header directory; see vision/README.md")
  endif()
  find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs OPTIONAL_COMPONENTS highgui)
  add_executable(detect_image src/detect_image.cpp)
  target_include_directories(detect_image PRIVATE include "${RKNN_INCLUDE_DIR}" ${OpenCV_INCLUDE_DIRS})
  target_compile_options(detect_image PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(detect_image PRIVATE ${OpenCV_LIBS} ${CMAKE_DL_LIBS})
  if(OpenCV_highgui_FOUND)
    target_compile_definitions(detect_image PRIVATE CAR_HAVE_HIGHGUI=1)
  endif()
  if(BUILD_TESTING)
    add_executable(capture_test tests/capture_test.cpp)
    target_include_directories(capture_test PRIVATE include ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(capture_test PRIVATE ${OpenCV_LIBS})
    target_compile_options(capture_test PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME capture_memory_and_cleanup COMMAND capture_test)
  endif()
endif()
